scripts.collect_report_assets

Gather the figures and tables most useful for the written report into one folder, report/, grouped by report section, with a README index explaining each file.

Everything here is COPIED from existing outputs (nothing is recomputed), apart from three small summary tables built from existing results: the data split, val vs test results for the chosen model, and an error-type breakdown. Re-run after any stage is rerun to refresh the folder.

Run with: .venv\Scripts\python.exe scripts\collect_report_assets.py

 81def expand(pattern: str) -> list[Path]:
 82    """Resolve 'path' or 'glob|N' (first N sorted matches) to existing files."""
 83    glob, _, n = pattern.partition("|")
 84    matches = sorted(REPO_ROOT.glob(glob))
 85    return matches[: int(n)] if n else matches

 88def split_table() -> pd.DataFrame:
 89    """Train/val/test sizes from the frozen configs/splits.json."""
 90    splits = json.loads((REPO_ROOT / "configs" / "splits.json").read_text(encoding="utf-8"))["splits"]
 91    manifest = pd.read_csv(load_paths()["manifest_csv"])
 92    apples = pd.read_csv(DERIVED / "apples_2d.csv")
 93    apples["tree_id"] = apples["image_id"].str.extract(r"^(R\d+U\d+)", expand=False)
 94    rows = []
 95    for name in ("train", "val", "test"):
 96        trees = splits[name]
 97        rows.append({
 98            "split": name,
 99            "orchard_rows": " ".join(str(r) for r in sorted({int(re.match(r"R(\d+)", t).group(1)) for t in trees})),
100            "trees": len(trees),
101            "trees_pct": round(100 * len(trees) / len(manifest), 1),
102            "labelled_images": int(manifest[manifest["tree_id"].isin(trees)]["has_labelme_json"].sum()),
103            "labelled_apples": int(apples["tree_id"].isin(trees).sum()),
104        })
105    return pd.DataFrame(rows)

108def val_vs_test_table() -> pd.DataFrame:
109    """Chosen model's val vs locked-test metrics, parsed from the two summary files."""
110    rows = []
111    for split, path in (("val", RESULTS / "model-v1-40train_val_summary.md"),
112                        ("test", RESULTS / "final_test_eval_summary.md")):
113        text = path.read_text(encoding="utf-8")
114        get = lambda key: float(re.search(rf"{key}=([-+\d.]+)", text).group(1))  # noqa: E731
115        rows.append({"split": split, "precision": get("precision"), "recall": get("recall"), "f1": get("F1"),
116                     "map50": get(r"mAP@0\.5"), "count_mae": get("MAE"), "count_bias": get("bias")})
117    return pd.DataFrame(rows)

120def error_breakdown() -> pd.DataFrame:
121    """Count of the chosen model's val errors by type, with mean confidence for false positives."""
122    errors = pd.read_csv(DERIVED / "error_analysis" / "model-v1-40train" / "errors.csv")
123    out = errors.groupby("type").agg(count=("type", "size"), mean_radius_px=("radius_px", "mean"),
124                                      mean_confidence=("score", "mean")).round(3).reset_index()
125    return out

128def main() -> int:
129    """Rebuild report/ from scratch and write its README index. Returns 0, or 1 if a source is missing."""
130    if OUT.exists():
131        shutil.rmtree(OUT)
132    index = ["# Report figures and tables\n",
133             "Collected by `scripts/collect_report_assets.py` from the pipeline's outputs. "
134             "All 3D numbers inherit the approximate depth scale and camera intrinsics "
135             "(see `docs/limitations.md`), and 3D error is a proxy (no physical ground truth).\n"]
136
137    missing = []
138    for section, items in SECTIONS.items():
139        (OUT / section).mkdir(parents=True)
140        index.append(f"\n## {section}\n")
141        for pattern, desc in items:
142            files = expand(pattern)
143            if not files:
144                missing.append(pattern)
145                continue
146            for src in files:
147                shutil.copy2(src, OUT / section / src.name)
148                index.append(f"- `{section}/{src.name}` — {desc}")
149
150    new_tables = [
151        ("01_data", "data_split.csv", split_table(), "Train/val/test split: orchard rows, trees, labelled images and apples per split."),
152        ("06_final_test", "val_vs_test.csv", val_vs_test_table(), "Chosen model on val vs the locked test set."),
153        ("04_error_analysis", "error_breakdown.csv", error_breakdown(), "Error counts by type, with mean size and confidence."),
154    ]
155    index.append("\n## Tables built for the report\n")
156    for section, name, df, desc in new_tables:
157        df.to_csv(OUT / section / name, index=False)
158        index.append(f"- `{section}/{name}` — {desc}")
159
160    (OUT / "README.md").write_text("\n".join(index) + "\n", encoding="utf-8")
161    n_files = sum(1 for p in OUT.rglob("*") if p.is_file())
162    print(f"Wrote {n_files} files to {OUT.relative_to(REPO_ROOT)}/")
163    if missing:
164        print("Missing sources (skipped):\n  " + "\n  ".join(missing))
165        return 1
166    return 0

170    raise SystemExit(main())
