src.evaluate

The one shared evaluation module (rule 7). Every method -- colour masking (Stage 6), YOLO (Stage 8), any hybrid (Stage 9) -- is scored by the same code here, on the same metrics, so results across methods are directly comparable instead of each method grading its own homework slightly differently.

Two kinds of apples flow through this module:

  • "ground truth" (gt): from labelme circles (via src/labels.py)
  • "predictions" (pred): from a detector or classical pipeline

Both are represented as Apple2D (image_id, cx, cy, radius_px, score).

 71def match_image(gt: list[Apple2D], pred: list[Apple2D], match_frac: float = 0.5) -> ImageMatchResult:
 72    """One-to-one Hungarian matching on centre distance, for apples in ONE image.
 73
 74    A match counts only if the distance is under `match_frac` * the
 75    GROUND-TRUTH apple's radius (the brief's rule: 0.5 * labelled radius).
 76    Hungarian assignment (scipy linear_sum_assignment) finds the
 77    globally-optimal one-to-one pairing minimising total distance, rather
 78    than each prediction greedily grabbing the nearest gt apple (which can
 79    produce avoidable bad pairings when two apples are close together).
 80    """
 81    image_id = gt[0].image_id if gt else (pred[0].image_id if pred else "")
 82    result = ImageMatchResult(image_id=image_id)
 83    if not gt or not pred:
 84        result.unmatched_gt = list(range(len(gt)))
 85        result.unmatched_pred = list(range(len(pred)))
 86        return result
 87
 88    cost = np.zeros((len(gt), len(pred)))
 89    for i, g in enumerate(gt):
 90        for j, p in enumerate(pred):
 91            cost[i, j] = float(np.hypot(g.cx - p.cx, g.cy - p.cy))
 92
 93    gt_idx, pred_idx = linear_sum_assignment(cost)
 94
 95    matched_gt, matched_pred = set(), set()
 96    for i, j in zip(gt_idx, pred_idx):
 97        dist = cost[i, j]
 98        threshold = match_frac * gt[i].radius_px
 99        if dist < threshold:
100            result.matched.append((int(i), int(j), float(dist)))
101            matched_gt.add(i)
102            matched_pred.add(j)
103
104    result.unmatched_gt = [i for i in range(len(gt)) if i not in matched_gt]
105    result.unmatched_pred = [j for j in range(len(pred)) if j not in matched_pred]
106    return result

109def match_all(gt: list[Apple2D], pred: list[Apple2D], match_frac: float = 0.5) -> dict[str, ImageMatchResult]:
110    """match_image, grouped by image_id, over a whole dataset."""
111    image_ids = sorted(set(a.image_id for a in gt) | set(a.image_id for a in pred))
112    results = {}
113    for image_id in image_ids:
114        g = [a for a in gt if a.image_id == image_id]
115        p = [a for a in pred if a.image_id == image_id]
116        results[image_id] = match_image(g, p, match_frac)
117    return results

122def precision_recall_f1(match_results: dict[str, ImageMatchResult]) -> dict:
123    """Standard detection metrics from match_all()'s output, pooled across every image:
124    precision = TP/(TP+FP), recall = TP/(TP+FN), F1 = harmonic mean of the two.
125    Returns {"tp", "fp", "fn", "precision", "recall", "f1"}.
126    """
127    tp = sum(len(r.matched) for r in match_results.values())
128    fp = sum(len(r.unmatched_pred) for r in match_results.values())
129    fn = sum(len(r.unmatched_gt) for r in match_results.values())
130    precision = tp / (tp + fp) if (tp + fp) > 0 else float("nan")
131    recall = tp / (tp + fn) if (tp + fn) > 0 else float("nan")
132    f1 = 2 * precision * recall / (precision + recall) if (precision + recall) > 0 else float("nan")
133    return {"tp": tp, "fp": fp, "fn": fn, "precision": precision, "recall": recall, "f1": f1}

136def _iou_square(a: Apple2D, b: Apple2D) -> float:
137    """IoU of the two apples' square detection boxes (2*radius side, centred on cx,cy) -- same
138    box convention as src/labels.py's YOLO detection export."""
139    ax0, ax1 = a.cx - a.radius_px, a.cx + a.radius_px
140    ay0, ay1 = a.cy - a.radius_px, a.cy + a.radius_px
141    bx0, bx1 = b.cx - b.radius_px, b.cx + b.radius_px
142    by0, by1 = b.cy - b.radius_px, b.cy + b.radius_px
143
144    ix0, iy0 = max(ax0, bx0), max(ay0, by0)
145    ix1, iy1 = min(ax1, bx1), min(ay1, by1)
146    iw, ih = max(0.0, ix1 - ix0), max(0.0, iy1 - iy0)
147    inter = iw * ih
148    area_a = (ax1 - ax0) * (ay1 - ay0)
149    area_b = (bx1 - bx0) * (by1 - by0)
150    union = area_a + area_b - inter
151    return inter / union if union > 0 else 0.0

154def average_precision_at_iou(gt: list[Apple2D], pred: list[Apple2D], iou_threshold: float = 0.5) -> float:
155    """Single-class (apple) Average Precision at a fixed IoU threshold, computed the standard
156    way: rank all predictions (across all images) by score, greedily assign each to its best
157    unmatched same-image GT box if IoU >= threshold, then take the area under the resulting
158    precision-recall curve (all-points / envelope interpolation).
159    """
160    if not gt:
161        return float("nan")
162    gt_by_image: dict[str, list[int]] = {}
163    for i, g in enumerate(gt):
164        gt_by_image.setdefault(g.image_id, []).append(i)
165    gt_taken = [False] * len(gt)
166
167    preds_sorted = sorted(enumerate(pred), key=lambda t: -t[1].score)
168    tp = np.zeros(len(preds_sorted))
169    fp = np.zeros(len(preds_sorted))
170
171    for rank, (_, p) in enumerate(preds_sorted):
172        best_iou, best_gt_i = 0.0, -1
173        for gi in gt_by_image.get(p.image_id, []):
174            if gt_taken[gi]:
175                continue
176            iou = _iou_square(gt[gi], p)
177            if iou > best_iou:
178                best_iou, best_gt_i = iou, gi
179        if best_iou >= iou_threshold:
180            tp[rank] = 1
181            gt_taken[best_gt_i] = True
182        else:
183            fp[rank] = 1
184
185    tp_cum, fp_cum = np.cumsum(tp), np.cumsum(fp)
186    recall = tp_cum / len(gt)
187    precision = tp_cum / np.maximum(tp_cum + fp_cum, 1e-12)
188
189    # envelope (precision is monotonically non-increasing as recall increases)
190    for i in range(len(precision) - 2, -1, -1):
191        precision[i] = max(precision[i], precision[i + 1])
192
193    recall = np.concatenate([[0.0], recall, [1.0]])
194    precision = np.concatenate([[precision[0] if len(precision) else 0.0], precision, [0.0]])
195    ap = float(np.sum((recall[1:] - recall[:-1]) * precision[1:]))
196    return ap

199# --- 2b. Apple-COUNT accuracy -------------------------------------------------
200#
201# A different question from precision/recall/F1: those score whether each
202# INDIVIDUAL apple was correctly localised. This scores whether the model
203# gets the TOTAL NUMBER of apples right per image/tree, even if some
204# individual matches are wrong (e.g. one detection drifted onto a
205# neighbouring apple while the count still came out correct) -- relevant
206# for yield-estimation use cases, not just a picking arm that needs each
207# apple's position.
208
209def count_accuracy(gt: list[Apple2D], pred: list[Apple2D]) -> dict:
210    """Per-image apple-count accuracy. Returns a dict with:
211      n_images, count_mae (mean absolute error in apples/image),
212      count_rmse, count_bias (mean signed error: pred - gt; positive =
213      systematic over-counting, negative = systematic under-counting),
214      count_correlation (Pearson r between per-image predicted and true
215      counts, NaN if either has zero variance), and per_image (a DataFrame
216      with one row per image: image_id, gt_count, pred_count, count_error).
217    """
218    image_ids = sorted(set(a.image_id for a in gt) | set(a.image_id for a in pred))
219    gt_counts = Counter(a.image_id for a in gt)
220    pred_counts = Counter(a.image_id for a in pred)
221
222    rows = []
223    for image_id in image_ids:
224        g, p = gt_counts.get(image_id, 0), pred_counts.get(image_id, 0)
225        rows.append({"image_id": image_id, "gt_count": g, "pred_count": p, "count_error": p - g})
226    per_image = pd.DataFrame(rows)
227
228    if per_image.empty:
229        return {"n_images": 0, "count_mae": float("nan"), "count_rmse": float("nan"),
230                "count_bias": float("nan"), "count_correlation": float("nan"), "per_image": per_image}
231
232    errors = per_image["count_error"].to_numpy(dtype=float)
233    gt_arr = per_image["gt_count"].to_numpy(dtype=float)
234    pred_arr = per_image["pred_count"].to_numpy(dtype=float)
235
236    if gt_arr.std() > 0 and pred_arr.std() > 0:
237        correlation = float(np.corrcoef(gt_arr, pred_arr)[0, 1])
238    else:
239        correlation = float("nan")  # undefined when every image has the same count on one side
240
241    return {
242        "n_images": len(image_ids),
243        "count_mae": float(np.mean(np.abs(errors))),
244        "count_rmse": float(np.sqrt(np.mean(errors**2))),
245        "count_bias": float(np.mean(errors)),
246        "count_correlation": correlation,
247        "per_image": per_image,
248    }

251# --- 3/4. Per-match 2D and 3D (proxy) error ----------------------------------

253def per_match_errors(
254    gt: list[Apple2D],
255    pred: list[Apple2D],
256    match_results: dict[str, ImageMatchResult],
257    depth_lookup: dict[str, np.ndarray],
258    intrinsics: Intrinsics,
259) -> pd.DataFrame:
260    """One row per matched apple pair: 2D pixel/cm error and the 3D proxy error.
261
262    3D error is a PROXY (item 4): there is no physical ground truth, so
263    "truth" here means the 3D FRONT-CENTRE POINT computed from the LABELLED
264    circle through the exact same geometry module used for the prediction
265    (src.geometry.apple_front_point_3d). This measures how much the
266    prediction's 2D centre disagrees with the label's 2D centre, expressed
267    in 3D -- not distance to a real, physically measured apple. State this
268    wherever the numbers are reported.
269    """
270    rows = []
271    gt_by_image: dict[str, list[Apple2D]] = {}
272    pred_by_image: dict[str, list[Apple2D]] = {}
273    for a in gt:
274        gt_by_image.setdefault(a.image_id, []).append(a)
275    for a in pred:
276        pred_by_image.setdefault(a.image_id, []).append(a)
277
278    for image_id, result in match_results.items():
279        depth_m = depth_lookup.get(image_id)
280        g_list = gt_by_image.get(image_id, [])
281        p_list = pred_by_image.get(image_id, [])
282
283        for gt_idx, pred_idx, dist_px in result.matched:
284            g, p = g_list[gt_idx], p_list[pred_idx]
285
286            row = {
287                "image_id": image_id,
288                "gt_index": gt_idx,
289                "pred_index": pred_idx,
290                "error_2d_px": dist_px,
291                "error_2d_cm": np.nan,
292                "error_3d_proxy_cm": np.nan,
293                "z_m": np.nan,
294                "distance_band": "unknown",
295            }
296
297            if depth_m is not None:
298                gt_pt = apple_front_point_3d(g.cx, g.cy, g.radius_px, depth_m, intrinsics)
299                pred_pt = apple_front_point_3d(p.cx, p.cy, p.radius_px, depth_m, intrinsics)
300
301                if gt_pt.is_valid:
302                    row["z_m"] = gt_pt.depth_sample.depth_m
303                    row["distance_band"] = distance_band(gt_pt.depth_sample.depth_m)
304
305                    du, dv = g.cx - p.cx, g.cy - p.cy
306                    z = gt_pt.depth_sample.depth_m
307                    dx, dy = du * z / intrinsics.fx, dv * z / intrinsics.fy
308                    row["error_2d_cm"] = 100 * float(np.hypot(dx, dy))
309
310                if gt_pt.is_valid and pred_pt.is_valid:
311                    row["error_3d_proxy_cm"] = 100 * float(np.linalg.norm(gt_pt.point_m - pred_pt.point_m))
312
313            rows.append(row)
314
315    return pd.DataFrame(rows)

320def summarise(
321    gt: list[Apple2D],
322    pred: list[Apple2D],
323    match_results: dict[str, ImageMatchResult],
324    match_df: pd.DataFrame,
325) -> dict:
326    """Overall + per-distance-band summary, ready to print as a markdown table."""
327    detection = precision_recall_f1(match_results)
328    detection["mAP@0.5"] = average_precision_at_iou(gt, pred, iou_threshold=0.5)
329
330    counting = count_accuracy(gt, pred)
331    counting_scalars = {k: v for k, v in counting.items() if k != "per_image"}
332
333    def _block(df: pd.DataFrame) -> dict:
334        """Median errors and within-2cm/3cm shares for one group of matched apples."""
335        if df.empty:
336            return {"n": 0}
337        valid_3d = df["error_3d_proxy_cm"].dropna()
338        return {
339            "n": len(df),
340            "median_error_2d_px": float(df["error_2d_px"].median()),
341            "median_error_2d_cm": float(df["error_2d_cm"].median(skipna=True)),
342            "median_error_3d_proxy_cm": float(valid_3d.median()) if len(valid_3d) else float("nan"),
343            "within_2cm_frac": float((valid_3d <= 2.0).mean()) if len(valid_3d) else float("nan"),
344            "within_3cm_frac": float((valid_3d <= 3.0).mean()) if len(valid_3d) else float("nan"),
345        }
346
347    summary = {"detection": detection, "counting": counting_scalars, "overall": _block(match_df),
348               "by_distance_band": {}}
349    # "out_of_range" (a valid depth, but outside every defined band -- this project's real
350    # data regularly falls below 0.4m due to the checkpoint-C1 depth-scale limitation) and
351    # "unknown" (no usable depth at all) must both be reported, not silently dropped --
352    # otherwise a result with every match out of range would print an empty, misleading table.
353    all_band_labels = [f"{lo:.1f}-{hi:.1f}m" for lo, hi in DISTANCE_BANDS] + ["out_of_range", "unknown"]
354    for band_label in all_band_labels:
355        sub = match_df[match_df["distance_band"] == band_label] if not match_df.empty else match_df
356        if len(sub):
357            summary["by_distance_band"][band_label] = _block(sub)
358    return summary

361def summary_to_markdown(summary: dict, method_name: str, split_name: str) -> str:
362    """Render summarise()'s output as a markdown report: detection metrics, counting
363    accuracy, overall error, and the per-distance-band table -- with the 3D-proxy
364    caveat spelled out every time, so a results file is never shown without it.
365    """
366    d = summary["detection"]
367    o = summary["overall"]
368    lines = [
369        f"### {method_name} on {split_name}",
370        "",
371        "**3D error is a proxy** (no physical ground truth): the distance between the 3D "
372        "front-centre point computed from the predicted circle and from the labelled circle, "
373        "through the same geometry module -- not distance to a physically measured apple.",
374        "",
375        f"- Detection: precision={d['precision']:.3f}, recall={d['recall']:.3f}, "
376        f"F1={d['f1']:.3f}, mAP@0.5={d['mAP@0.5']:.3f} (TP={d['tp']}, FP={d['fp']}, FN={d['fn']})",
377    ]
378    c = summary.get("counting")
379    if c and c.get("n_images", 0):
380        corr_str = f"{c['count_correlation']:.3f}" if np.isfinite(c["count_correlation"]) else "n/a"
381        lines.append(
382            f"- **Apple count per image** (how close is the TOTAL count, not individual "
383            f"matches -- {c['n_images']} images): MAE={c['count_mae']:.2f} apples/image, "
384            f"RMSE={c['count_rmse']:.2f}, bias={c['count_bias']:+.2f} "
385            f"({'over' if c['count_bias'] > 0 else 'under' if c['count_bias'] < 0 else 'no'}-counting "
386            f"on average), correlation={corr_str}"
387        )
388    if o.get("n", 0):
389        lines.append(
390            f"- Overall ({o['n']} matched apples): median 2D error = {o['median_error_2d_px']:.2f}px "
391            f"({o['median_error_2d_cm']:.2f}cm); median 3D proxy error = "
392            f"{o['median_error_3d_proxy_cm']:.2f}cm; within 2cm = {100*o['within_2cm_frac']:.1f}%; "
393            f"within 3cm = {100*o['within_3cm_frac']:.1f}%"
394        )
395    else:
396        lines.append("- Overall: no matched apples")
397
398    lines.append("\n| Distance band | n | median 2D err (cm) | median 3D proxy err (cm) | within 2cm | within 3cm |")
399    lines.append("|---|---|---|---|---|---|")
400    for band, b in summary["by_distance_band"].items():
401        if b.get("n", 0):
402            lines.append(
403                f"| {band} | {b['n']} | {b['median_error_2d_cm']:.2f} | "
404                f"{b['median_error_3d_proxy_cm']:.2f} | {100*b['within_2cm_frac']:.1f}% | "
405                f"{100*b['within_3cm_frac']:.1f}% |"
406            )
407    return "\n".join(lines)

 75    A match counts only if the distance is under `match_frac` * the
 76    GROUND-TRUTH apple's radius (the brief's rule: 0.5 * labelled radius).
 77    Hungarian assignment (scipy linear_sum_assignment) finds the
 78    globally-optimal one-to-one pairing minimising total distance, rather
 79    than each prediction greedily grabbing the nearest gt apple (which can
 80    produce avoidable bad pairings when two apples are close together).
 81    """
 82    image_id = gt[0].image_id if gt else (pred[0].image_id if pred else "")
 83    result = ImageMatchResult(image_id=image_id)
 84    if not gt or not pred:
 85        result.unmatched_gt = list(range(len(gt)))
 86        result.unmatched_pred = list(range(len(pred)))
 87        return result
 88
 89    cost = np.zeros((len(gt), len(pred)))
 90    for i, g in enumerate(gt):
 91        for j, p in enumerate(pred):
 92            cost[i, j] = float(np.hypot(g.cx - p.cx, g.cy - p.cy))
 93
 94    gt_idx, pred_idx = linear_sum_assignment(cost)
 95
 96    matched_gt, matched_pred = set(), set()
 97    for i, j in zip(gt_idx, pred_idx):
 98        dist = cost[i, j]
 99        threshold = match_frac * gt[i].radius_px
100        if dist < threshold:
101            result.matched.append((int(i), int(j), float(dist)))
102            matched_gt.add(i)
103            matched_pred.add(j)
104
105    result.unmatched_gt = [i for i in range(len(gt)) if i not in matched_gt]
106    result.unmatched_pred = [j for j in range(len(pred)) if j not in matched_pred]
107    return result

def precision_recall_f1(match_results: dict[str, ImageMatchResult]) -> dict: 
123def precision_recall_f1(match_results: dict[str, ImageMatchResult]) -> dict:
124    """Standard detection metrics from match_all()'s output, pooled across every image:
125    precision = TP/(TP+FP), recall = TP/(TP+FN), F1 = harmonic mean of the two.
126    Returns {"tp", "fp", "fn", "precision", "recall", "f1"}.
127    """
128    tp = sum(len(r.matched) for r in match_results.values())
129    fp = sum(len(r.unmatched_pred) for r in match_results.values())
130    fn = sum(len(r.unmatched_gt) for r in match_results.values())
131    precision = tp / (tp + fp) if (tp + fp) > 0 else float("nan")
132    recall = tp / (tp + fn) if (tp + fn) > 0 else float("nan")
133    f1 = 2 * precision * recall / (precision + recall) if (precision + recall) > 0 else float("nan")
134    return {"tp": tp, "fp": fp, "fn": fn, "precision": precision, "recall": recall, "f1": f1}

def average_precision_at_iou( gt: list[Apple2D], pred: list[Apple2D], iou_threshold: float = 0.5) -> float: 
155def average_precision_at_iou(gt: list[Apple2D], pred: list[Apple2D], iou_threshold: float = 0.5) -> float:
156    """Single-class (apple) Average Precision at a fixed IoU threshold, computed the standard
157    way: rank all predictions (across all images) by score, greedily assign each to its best
158    unmatched same-image GT box if IoU >= threshold, then take the area under the resulting
159    precision-recall curve (all-points / envelope interpolation).
160    """
161    if not gt:
162        return float("nan")
163    gt_by_image: dict[str, list[int]] = {}
164    for i, g in enumerate(gt):
165        gt_by_image.setdefault(g.image_id, []).append(i)
166    gt_taken = [False] * len(gt)
167
168    preds_sorted = sorted(enumerate(pred), key=lambda t: -t[1].score)
169    tp = np.zeros(len(preds_sorted))
170    fp = np.zeros(len(preds_sorted))
171
172    for rank, (_, p) in enumerate(preds_sorted):
173        best_iou, best_gt_i = 0.0, -1
174        for gi in gt_by_image.get(p.image_id, []):
175            if gt_taken[gi]:
176                continue
177            iou = _iou_square(gt[gi], p)
178            if iou > best_iou:
179                best_iou, best_gt_i = iou, gi
180        if best_iou >= iou_threshold:
181            tp[rank] = 1
182            gt_taken[best_gt_i] = True
183        else:
184            fp[rank] = 1
185
186    tp_cum, fp_cum = np.cumsum(tp), np.cumsum(fp)
187    recall = tp_cum / len(gt)
188    precision = tp_cum / np.maximum(tp_cum + fp_cum, 1e-12)
189
190    # envelope (precision is monotonically non-increasing as recall increases)
191    for i in range(len(precision) - 2, -1, -1):
192        precision[i] = max(precision[i], precision[i + 1])
193
194    recall = np.concatenate([[0.0], recall, [1.0]])
195    precision = np.concatenate([[precision[0] if len(precision) else 0.0], precision, [0.0]])
196    ap = float(np.sum((recall[1:] - recall[:-1]) * precision[1:]))
197    return ap

def count_accuracy(gt: list[Apple2D], pred: list[Apple2D]) -> dict: 
210def count_accuracy(gt: list[Apple2D], pred: list[Apple2D]) -> dict:
211    """Per-image apple-count accuracy. Returns a dict with:
212      n_images, count_mae (mean absolute error in apples/image),
213      count_rmse, count_bias (mean signed error: pred - gt; positive =
214      systematic over-counting, negative = systematic under-counting),
215      count_correlation (Pearson r between per-image predicted and true
216      counts, NaN if either has zero variance), and per_image (a DataFrame
217      with one row per image: image_id, gt_count, pred_count, count_error).
218    """
219    image_ids = sorted(set(a.image_id for a in gt) | set(a.image_id for a in pred))
220    gt_counts = Counter(a.image_id for a in gt)
221    pred_counts = Counter(a.image_id for a in pred)
222
223    rows = []
224    for image_id in image_ids:
225        g, p = gt_counts.get(image_id, 0), pred_counts.get(image_id, 0)
226        rows.append({"image_id": image_id, "gt_count": g, "pred_count": p, "count_error": p - g})
227    per_image = pd.DataFrame(rows)
228
229    if per_image.empty:
230        return {"n_images": 0, "count_mae": float("nan"), "count_rmse": float("nan"),
231                "count_bias": float("nan"), "count_correlation": float("nan"), "per_image": per_image}
232
233    errors = per_image["count_error"].to_numpy(dtype=float)
234    gt_arr = per_image["gt_count"].to_numpy(dtype=float)
235    pred_arr = per_image["pred_count"].to_numpy(dtype=float)
236
237    if gt_arr.std() > 0 and pred_arr.std() > 0:
238        correlation = float(np.corrcoef(gt_arr, pred_arr)[0, 1])
239    else:
240        correlation = float("nan")  # undefined when every image has the same count on one side
241
242    return {
243        "n_images": len(image_ids),
244        "count_mae": float(np.mean(np.abs(errors))),
245        "count_rmse": float(np.sqrt(np.mean(errors**2))),
246        "count_bias": float(np.mean(errors)),
247        "count_correlation": correlation,
248        "per_image": per_image,
249    }
