src.geometry

The one shared module for turning a detected apple (2D pixel centre) into a 3D position. Every method in this project (classical colour masking, YOLO, any hybrid) must go through these functions, so results are directly comparable (rule 7: one evaluation code path per concern).

What this module outputs: the 3D FRONT-CENTRE POINT of each apple -- the point on the apple's front (camera-facing) surface at its 2D centre pixel. Depth is used for exactly one thing: to place that one visible point in 3D (robust depth sample at the centre, then back-project).

Coordinate convention (camera frame, metres): X: right (increases with pixel column u) Y: down (increases with pixel row v) Z: forward, away from the camera, along the optical axis This matches the usual pinhole-camera / OpenCV / RealSense convention: the camera sits at the origin looking down +Z, with no coordinate flips.

Pixel convention: u is the column (x, 0 = left edge), v is the row (y, 0 = top edge), matching image-array indexing [row, col] = [v, u].

Units: pixel coordinates and radii are in pixels unless stated "_m" for metres. Depth arrays passed into this module must already be in METRES (converting from whatever raw format a depth file uses, e.g. the 8-bit raw-as-mm approximation documented in configs/camera.yaml, is the caller's job).

 48def backproject(u: float, v: float, z: float, intrinsics: Intrinsics) -> np.ndarray:
 49    """Convert one pixel + depth into a camera-frame 3D point.
 50
 51    Inputs:
 52      u, v: pixel coordinates (column, row), in pixels.
 53      z: depth at that pixel, in METRES (the distance along the optical
 54         axis from the camera to the point -- not the straight-line
 55         distance from the camera to the point).
 56      intrinsics: fx, fy, cx, cy in pixels.
 57    Output:
 58      numpy array [X, Y, Z] in metres, camera frame (see module docstring
 59      for the axis convention).
 60
 61    Derivation: a pinhole camera maps a 3D point (X, Y, Z) to pixel
 62    (u, v) = (fx * X/Z + cx, fy * Y/Z + cy). Solving for X, Y given a known
 63    Z (from the depth sensor) inverts that:
 64      X = (u - cx) * Z / fx
 65      Y = (v - cy) * Z / fy
 66    """
 67    x = (u - intrinsics.cx) * z / intrinsics.fx
 68    y = (v - intrinsics.cy) * z / intrinsics.fy
 69    return np.array([x, y, z], dtype=float)

 81def robust_depth(
 82    depth_m: np.ndarray,
 83    cx: float,
 84    cy: float,
 85    radius_px: float,
 86    inner_frac: float = 0.5,
 87    min_valid_pixels: int = 5,
 88    outlier_mad_k: float = 3.5,
 89) -> DepthSample:
 90    """Estimate one reliable depth value for an apple from a patch of pixels.
 91
 92    Rather than trusting the single pixel at the labelled/detected centre
 93    (easy to land on a gap between leaves, a noisy pixel, or the image
 94    border), this samples a small disc in the MIDDLE of the apple -- radius
 95    `inner_frac * radius_px` around (cx, cy) -- where the surface is most
 96    likely to be the apple itself and not background or occluding leaves,
 97    then takes the median after dropping zeros (the sensor's "no depth"
 98    sentinel) and statistical outliers.
 99
100    Inputs:
101      depth_m: full depth image in METRES, shape (H, W). 0 or NaN = invalid pixel.
102      cx, cy: apple centre in pixels.
103      radius_px: 2D circle radius in pixels (from the detector or label).
104      inner_frac: how much of the radius to sample from (0.5 = inner half).
105      min_valid_pixels: below this many valid pixels, treat the estimate as unreliable.
106      outlier_mad_k: pixels more than this many median-absolute-deviations
107        from the patch median are dropped as outliers (e.g. a stray
108        background pixel that leaked into the inner disc).
109    Output:
110      DepthSample(depth_m, valid_pixel_count, is_valid, reason). depth_m is
111      None and reason is set whenever is_valid is False -- callers must
112      never silently treat a missing depth as zero (rule 4, Stage 4 item 4).
113    """
114    h, w = depth_m.shape
115    inner_radius = max(1.0, inner_frac * radius_px)
116
117    v0, v1 = max(0, int(cy - inner_radius)), min(h, int(cy + inner_radius) + 1)
118    u0, u1 = max(0, int(cx - inner_radius)), min(w, int(cx + inner_radius) + 1)
119    if v1 <= v0 or u1 <= u0:
120        return DepthSample(None, 0, False, reason="sample window falls entirely outside the image")
121
122    vv, uu = np.mgrid[v0:v1, u0:u1]
123    dist = np.hypot(uu - cx, vv - cy)
124    disc_mask = dist <= inner_radius
125    patch = depth_m[v0:v1, u0:u1]
126
127    valid_mask = disc_mask & np.isfinite(patch) & (patch > 0)
128    values = patch[valid_mask]
129
130    if values.size == 0:
131        return DepthSample(None, 0, False, reason="no valid (non-zero, finite) depth pixels in the inner disc")
132
133    median = float(np.median(values))
134    mad = float(np.median(np.abs(values - median)))
135    if mad > 0:
136        keep = np.abs(values - median) <= outlier_mad_k * mad * 1.4826  # 1.4826 scales MAD to a std-dev estimate
137        values = values[keep]
138
139    if values.size == 0:
140        return DepthSample(None, 0, False, reason="all pixels rejected as outliers")
141
142    if values.size < min_valid_pixels:
143        return DepthSample(float(np.median(values)), int(values.size), False,
144                            reason=f"only {values.size} valid pixels, below min_valid_pixels={min_valid_pixels}")
145
146    return DepthSample(float(np.median(values)), int(values.size), True, reason=None)

159def apple_front_point_3d(
160    u: float,
161    v: float,
162    radius_px: float,
163    depth_m: np.ndarray,
164    intrinsics: Intrinsics,
165    **robust_depth_kwargs,
166) -> FrontPointResult:
167    """Compute the 3D front-centre point of an apple from its 2D detection.
168
169    Samples a robust depth Z around the 2D centre (robust_depth), then
170    back-projects (u, v, Z) to camera-frame XYZ. The output is that front
171    surface point itself, with nothing added to it.
172
173    Inputs:
174      u, v: 2D apple centre, in pixels.
175      radius_px: 2D radius in pixels, used only to size robust_depth's
176        sampling disc.
177      depth_m: full depth image in METRES, shape (H, W).
178      intrinsics: fx, fy, cx, cy in pixels.
179    Output:
180      FrontPointResult; point_m is [X, Y, Z] in metres (camera frame), or
181      None (with `reason` set) whenever depth is missing or too sparse to
182      trust -- never a silent zero.
183    """
184    depth_sample = robust_depth(depth_m, u, v, radius_px, **robust_depth_kwargs)
185    if not depth_sample.is_valid or depth_sample.depth_m is None:
186        return FrontPointResult(None, depth_sample, False, reason=depth_sample.reason)
187
188    point = backproject(u, v, depth_sample.depth_m, intrinsics)
189    return FrontPointResult(point, depth_sample, True, reason=None)
